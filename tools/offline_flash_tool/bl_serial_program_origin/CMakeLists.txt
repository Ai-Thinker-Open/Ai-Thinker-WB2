set(MSC_MEDIA 2)

set(BSP_COMMON_DIR ${CMAKE_SOURCE_DIR}/bsp/bsp_common)
set(TARGET_REQUIRED_LIBS usb_stack freertos fatfs shell)

if(MSC_MEDIA EQUAL 0)
set(TARGET_REQUIRED_PRIVATE_OPTIONS -DMSC_MEDIA=MSC_RAM)
elseif(MSC_MEDIA EQUAL 1)
set(TARGET_REQUIRED_PRIVATE_OPTIONS -DMSC_MEDIA=MSC_PSRAM -DPSRAM_BLOCK_COUNT=256 -DPSRAM_START_ADDR=0xa8000000)
set(TARGET_REQUIRED_SRCS ${BSP_COMMON_DIR}/fatfs/fatfs_psram.c)
elseif(MSC_MEDIA EQUAL 2)
set(TARGET_REQUIRED_PRIVATE_OPTIONS -DMSC_MEDIA=MSC_FLASH -DFLASH_BLOCK_COUNT=256 -DFLASH_START_ADDR=0x60000)
set(TARGET_REQUIRED_SRCS ${BSP_COMMON_DIR}/fatfs/fatfs_flash.c)
elseif(MSC_MEDIA EQUAL 3)
set(TARGET_REQUIRED_PRIVATE_OPTIONS -DMSC_MEDIA=MSC_SPI_SD)
set(TARGET_REQUIRED_SRCS ${BSP_COMMON_DIR}/sd_card/bsp_spi_sdcard.c)
elseif(MSC_MEDIA EQUAL 4)
set(TARGET_REQUIRED_PRIVATE_OPTIONS -DMSC_MEDIA=MSC_SDH)
set(TARGET_REQUIRED_SRCS ${BSP_COMMON_DIR}/sd_card/bsp_sdio_sdcard.c)
endif()
list(APPEND TARGET_REQUIRED_PRIVATE_OPTIONS -DCONFIG_USBDEV_MSC_THREAD -DROOT_PATH="2:")
set(TARGET_REQUIRED_PRIVATE_INCLUDE ${BSP_COMMON_DIR}/sd_card)

list(APPEND GLOBAL_C_FLAGS -DFF_MAX_SS=4096 -fshort-enums)
list(APPEND TARGET_REQUIRED_SRCS bootrom_isp.c)
set(mains main.c)
generate_bin()
